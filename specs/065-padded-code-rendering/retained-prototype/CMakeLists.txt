cmake_minimum_required(VERSION 3.21)
project(retained_glyph_gate LANGUAGES CXX)
if(NOT QMARKDOWN_STAGE)
    message(FATAL_ERROR "Set QMARKDOWN_STAGE to the patched staged source tree")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
set(QMARKDOWN_WITH_MATH OFF CACHE BOOL "" FORCE)
add_subdirectory("${QMARKDOWN_STAGE}" staged)
add_executable(retained-glyph-gate main.cpp)
target_include_directories(retained-glyph-gate PRIVATE "${QMARKDOWN_STAGE}/src/QMarkdown")
target_link_libraries(retained-glyph-gate PRIVATE QMarkdown Qt6::Quick)
target_compile_features(retained-glyph-gate PRIVATE cxx_std_17)

add_executable(retained-glyph-diagnostic diagnostic.cpp)
target_include_directories(retained-glyph-diagnostic PRIVATE "${QMARKDOWN_STAGE}/src/QMarkdown")
target_link_libraries(retained-glyph-diagnostic PRIVATE QMarkdown Qt6::Quick)
target_compile_features(retained-glyph-diagnostic PRIVATE cxx_std_17)

add_executable(whole-run-gate whole_run.cpp)
target_include_directories(whole-run-gate PRIVATE "${QMARKDOWN_STAGE}/src/QMarkdown")
target_link_libraries(whole-run-gate PRIVATE QMarkdown Qt6::Quick)
target_compile_features(whole-run-gate PRIVATE cxx_std_17)

add_executable(scenegraph-run-gate scenegraph.cpp)
target_include_directories(scenegraph-run-gate PRIVATE "${QMARKDOWN_STAGE}/src/QMarkdown")
target_link_libraries(scenegraph-run-gate PRIVATE QMarkdown Qt6::Quick)
target_compile_features(scenegraph-run-gate PRIVATE cxx_std_17)

# Optional existing public Vulkan headers for device identity in a minimal SDK.
if(SCENEGRAPH_VULKAN_INCLUDE_DIR)
    target_include_directories(scenegraph-run-gate PRIVATE "${SCENEGRAPH_VULKAN_INCLUDE_DIR}")
endif()
