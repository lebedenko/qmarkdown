cmake_minimum_required(VERSION 3.21)
project(blur_probe LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick Test)
set(QMARKDOWN_WITH_MATH OFF CACHE BOOL "" FORCE)
if(QMARKDOWN_STAGE)
    add_subdirectory("${QMARKDOWN_STAGE}" qmarkdown)
else()
    get_filename_component(QMARKDOWN_STAGE "../../.." ABSOLUTE)
    add_subdirectory("${QMARKDOWN_STAGE}" qmarkdown)
endif()
add_executable(blur-probe main.cpp)
target_include_directories(blur-probe PRIVATE ../../../src/QMarkdown)
target_link_libraries(blur-probe PRIVATE QMarkdown Qt6::Quick Qt6::Test)
target_compile_features(blur-probe PRIVATE cxx_std_17)

if(EXISTS "${QMARKDOWN_STAGE}/blur-source.json")
    add_executable(blur-motion motion.cpp)
    target_include_directories(blur-motion PRIVATE "${QMARKDOWN_STAGE}/src/QMarkdown")
    target_link_libraries(blur-motion PRIVATE QMarkdown Qt6::Quick Qt6::Test)
    target_compile_features(blur-motion PRIVATE cxx_std_17)
    target_compile_definitions(blur-motion PRIVATE BLUR_MANIFEST="${QMARKDOWN_STAGE}/blur-source.json")
endif()
