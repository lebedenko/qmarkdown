cmake_minimum_required(VERSION 3.21)
project(QMarkdownStreamingMeasurements LANGUAGES C CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(QMARKDOWN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick Network)
add_subdirectory(../../.. qmarkdown)
qt_add_executable(streaming-measurements main.cpp)
target_include_directories(streaming-measurements PRIVATE ../../../src/QMarkdown)
target_link_libraries(streaming-measurements PRIVATE QMarkdown::QMarkdown)
