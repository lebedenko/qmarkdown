cmake_minimum_required(VERSION 3.21)
project(scale_contract_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(PROBE_QT)
    find_package(Qt6 REQUIRED COMPONENTS Gui)
    if(NOT Qt6_VERSION VERSION_EQUAL EXPECTED_QT)
        message(FATAL_ERROR "Exact Qt mismatch")
    endif()
    add_executable(qt-probe probe.cpp)
    target_compile_definitions(qt-probe PRIVATE PROBE_QT)
    target_link_libraries(qt-probe PRIVATE Qt6::Gui)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2=2.32.74)
    pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3=3.4.18)
    add_executable(sdl2-probe probe.cpp)
    target_link_libraries(sdl2-probe PRIVATE PkgConfig::SDL2)
    add_executable(sdl3-probe probe.cpp)
    target_compile_definitions(sdl3-probe PRIVATE PROBE_SDL3)
    target_link_libraries(sdl3-probe PRIVATE PkgConfig::SDL3)
    add_executable(test-mapping test_mapping.cpp)
    enable_testing()
    add_test(NAME mapping COMMAND test-mapping)
endif()
