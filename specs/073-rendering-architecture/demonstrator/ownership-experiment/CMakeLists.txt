cmake_minimum_required(VERSION 3.21)
project(ownership_verification LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET freetype2 harfbuzz icu-uc icu-i18n libpng fontconfig zlib)
file(STRINGS "${SKIA}/out/comparison/obj/skia.ninja" DEFINES REGEX "^defines = ")
string(REPLACE "defines = " "" DEFINES "${DEFINES}")
separate_arguments(DEFINES UNIX_COMMAND "${DEFINES}")
file(GLOB LIBS "${SKIA}/out/comparison/*.a")
add_executable(phase-trace trace.cpp)
add_executable(perf-geometry ../boundary-experiment/main.cpp)
add_executable(ownership-tests tests.cpp)
foreach(target phase-trace perf-geometry ownership-tests)
    target_include_directories(${target} PRIVATE "${SKIA}")
    target_compile_options(${target} PRIVATE ${DEFINES})
    target_compile_definitions(${target} PRIVATE FRAGMENT_PATCHED)
    target_link_libraries(${target} PRIVATE "-Wl,--start-group" ${LIBS} "-Wl,--end-group" PkgConfig::DEPS dl pthread)
endforeach()
if(COUNTERS)
    target_compile_definitions(phase-trace PRIVATE PHASE_COUNTERS)
endif()
