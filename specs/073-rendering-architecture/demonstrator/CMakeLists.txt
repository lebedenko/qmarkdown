cmake_minimum_required(VERSION 3.21)
project(rendering_demonstrator LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
if(NOT EXISTS "${PREPARED_STAGE}/demo-source.json")
    message(FATAL_ERROR "Run stage.py into a fresh directory and set PREPARED_STAGE")
endif()
set(QMARKDOWN_WITH_MATH OFF CACHE BOOL "" FORCE)
add_subdirectory("${PREPARED_STAGE}" prepared)
add_executable(qt-compare qt.cpp)
target_include_directories(qt-compare PRIVATE "${PREPARED_STAGE}/src/QMarkdown")
target_link_libraries(qt-compare PRIVATE QMarkdown Qt6::Quick)
target_compile_features(qt-compare PRIVATE cxx_std_17)
add_executable(qt-selection selection.cpp)
target_include_directories(qt-selection PRIVATE "${PREPARED_STAGE}/src/QMarkdown")
target_link_libraries(qt-selection PRIVATE QMarkdown Qt6::Quick)
target_compile_features(qt-selection PRIVATE cxx_std_17)
add_executable(qt-lifecycle lifecycle.cpp)
target_link_libraries(qt-lifecycle PRIVATE Qt6::Quick)
target_compile_features(qt-lifecycle PRIVATE cxx_std_17)
