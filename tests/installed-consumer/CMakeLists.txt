cmake_minimum_required(VERSION 3.21)
project(QMarkdownInstalledConsumer LANGUAGES CXX)
set(QMARKDOWN_REQUEST_VERSION "1.0" CACHE STRING "Requested package version")
set(QMARKDOWN_PACKAGE_PREFIX "" CACHE PATH "Restrict QMarkdown discovery to this package prefix")
set(qmarkdown_find_options)
if(QMARKDOWN_PACKAGE_PREFIX)
    list(APPEND qmarkdown_find_options PATHS "${QMARKDOWN_PACKAGE_PREFIX}" NO_DEFAULT_PATH)
endif()
option(QMARKDOWN_REQUEST_EXACT "Require exact package version" OFF)
if(QMARKDOWN_REQUEST_EXACT)
    find_package(QMarkdown ${QMARKDOWN_REQUEST_VERSION} EXACT CONFIG REQUIRED ${qmarkdown_find_options})
else()
    find_package(QMarkdown ${QMARKDOWN_REQUEST_VERSION} CONFIG REQUIRED ${qmarkdown_find_options})
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
option(QMARKDOWN_STYLING_CONSUMER "Verify the 1.1 styling API" OFF)
option(QMARKDOWN_ACTIONS_CONSUMER "Verify the 1.2 code action API" OFF)
option(QMARKDOWN_TASKS_CONSUMER "Verify the 1.3 task-list API" OFF)
option(QMARKDOWN_STRIKE_CONSUMER "Verify the 1.4 strikethrough API" OFF)
option(QMARKDOWN_AUTOLINK_CONSUMER "Verify the 1.5 extended-autolink API" OFF)
option(QMARKDOWN_GFM_CONSUMER "Verify the 1.6 GFM/table API" OFF)
option(QMARKDOWN_SELECTION_CONSUMER "Verify the 1.8 selection API" OFF)
option(QMARKDOWN_RENDERERS_CONSUMER "Verify the custom fenced renderer API" OFF)
option(QMARKDOWN_READING_CONSUMER "Verify the 1.9 reading API" OFF)
option(QMARKDOWN_KEYBOARD_CONSUMER "Verify the 1.10 keyboard/accessibility API" OFF)
if(QMARKDOWN_KEYBOARD_CONSUMER)
    set(CMAKE_AUTOMOC ON)
    add_executable(qmarkdown-consumer keyboard.cpp)
elseif(QMARKDOWN_READING_CONSUMER)
    add_executable(qmarkdown-consumer reading.cpp)
elseif(QMARKDOWN_RENDERERS_CONSUMER)
    add_executable(qmarkdown-consumer renderers.cpp)
elseif(QMARKDOWN_SELECTION_CONSUMER)
    add_executable(qmarkdown-consumer selection.cpp)
elseif(QMARKDOWN_GFM_CONSUMER)
    add_executable(qmarkdown-consumer gfm.cpp)
elseif(QMARKDOWN_AUTOLINK_CONSUMER)
    add_executable(qmarkdown-consumer autolinks.cpp)
elseif(QMARKDOWN_STRIKE_CONSUMER)
    add_executable(qmarkdown-consumer strikethrough.cpp)
elseif(QMARKDOWN_TASKS_CONSUMER)
    add_executable(qmarkdown-consumer tasks.cpp)
elseif(QMARKDOWN_ACTIONS_CONSUMER)
    add_executable(qmarkdown-consumer actions.cpp)
elseif(QMARKDOWN_STYLING_CONSUMER)
    add_executable(qmarkdown-consumer styling.cpp)
else()
    add_executable(qmarkdown-consumer main.cpp)
endif()
target_compile_features(qmarkdown-consumer PRIVATE cxx_std_17)
target_link_libraries(qmarkdown-consumer PRIVATE Qt6::Test)
option(QMARKDOWN_CONSUMER_LINK_MODULE "Link the backing library" ON)
if(QMARKDOWN_CONSUMER_LINK_MODULE)
    target_link_libraries(qmarkdown-consumer PRIVATE QMarkdown::QMarkdown)
else()
    target_link_libraries(qmarkdown-consumer PRIVATE Qt6::Quick Qt6::Qml)
endif()
if(TARGET QMarkdown::QMarkdownPlugin)
    target_link_libraries(qmarkdown-consumer PRIVATE QMarkdown::QMarkdownPlugin)
    target_compile_definitions(qmarkdown-consumer PRIVATE QMARKDOWN_STATIC)
endif()
