cmake_minimum_required(VERSION 3.21)
project(QMarkdownInstalledConsumer LANGUAGES CXX)
set(QMARKDOWN_REQUEST_VERSION "" CACHE STRING "Requested package version")
if(NOT QMARKDOWN_REQUEST_VERSION)
    message(FATAL_ERROR "Set QMARKDOWN_REQUEST_VERSION to the current project version (verify-packaging.py supplies it)")
endif()
set(QMARKDOWN_PACKAGE_PREFIX "" CACHE PATH "Restrict QMarkdown discovery to this package prefix")
set(qmarkdown_find_options)
if(QMARKDOWN_PACKAGE_PREFIX)
    list(APPEND qmarkdown_find_options PATHS "${QMARKDOWN_PACKAGE_PREFIX}" NO_DEFAULT_PATH)
endif()
option(QMARKDOWN_REQUEST_EXACT "Require exact package version" ON)
if(QMARKDOWN_REQUEST_EXACT)
    find_package(QMarkdown ${QMARKDOWN_REQUEST_VERSION} EXACT CONFIG REQUIRED ${qmarkdown_find_options})
else()
    find_package(QMarkdown ${QMARKDOWN_REQUEST_VERSION} CONFIG REQUIRED ${qmarkdown_find_options})
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
set(CMAKE_AUTOMOC ON)
set(QMARKDOWN_EXPECT_SYNTAX "0" CACHE STRING "Expected syntax backend")
set(QMARKDOWN_EXPECT_MATH "1" CACHE STRING "Expected math backend")
enable_testing()
get_target_property(module_type QMarkdown::QMarkdown TYPE)
set(modes linked)
if(module_type STREQUAL "SHARED_LIBRARY")
    list(APPEND modes plugin)
endif()
foreach(mode IN LISTS modes)
    foreach(fixture main styling actions tasks strikethrough autolinks gfm selection renderers reading math footnotes providers syntax keyboard)
        set(target consumer-${fixture}-${mode})
        add_executable(${target} ${fixture}.cpp)
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_link_libraries(${target} PRIVATE Qt6::Test Qt6::Quick Qt6::Qml)
        target_compile_definitions(${target} PRIVATE
            QMARKDOWN_EXPECT_SYNTAX=${QMARKDOWN_EXPECT_SYNTAX}
            QMARKDOWN_EXPECT_MATH=${QMARKDOWN_EXPECT_MATH})
        if(mode STREQUAL "linked")
            target_link_libraries(${target} PRIVATE QMarkdown::QMarkdown)
        endif()
        if(TARGET QMarkdown::QMarkdownPlugin)
            target_link_libraries(${target} PRIVATE QMarkdown::QMarkdownPlugin)
            target_compile_definitions(${target} PRIVATE QMARKDOWN_STATIC)
        endif()
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
endforeach()
